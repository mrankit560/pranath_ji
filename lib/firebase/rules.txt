// =========================================================================
// FIRESTORE SECURITY RULES (firestore.rules)
// Shri Prannath Ji Digital Gyan Portal
// =========================================================================

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Helper functions
    function isAuthenticated() {
      return request.auth != null;
    }
    
    function isAdmin() {
      return isAuthenticated() && 
        (request.auth.token.role == 'admin' || 
         request.auth.token.role == 'superadmin' || 
         request.auth.token.email in ['sadhaulidham@gmail.com', 'admin@sadhaulidham.org']);
    }
    
    function isOwner(userId) {
      return isAuthenticated() && request.auth.uid == userId;
    }

    // Public readable spiritual collections, write-restricted to admins
    match /scriptures/{scriptureId} {
      allow read: if true;
      allow write: if isAdmin();
    }
    
    match /books/{bookId} {
      allow read: if resource.data.published == true || isAdmin();
      allow write: if isAdmin();
    }
    
    match /videos/{videoId} {
      allow read: if resource.data.published == true || isAdmin();
      allow write: if isAdmin();
    }
    
    match /audio/{audioId} {
      allow read: if resource.data.published == true || isAdmin();
      allow write: if isAdmin();
    }
    
    match /events/{eventId} {
      allow read: if true;
      allow write: if isAdmin();
    }
    
    match /articles/{articleId} {
      allow read: if resource.data.status == 'published' || isAdmin();
      allow write: if isAdmin();
    }
    
    match /spiritualTopics/{topicId} {
      allow read: if true;
      allow write: if isAdmin();
    }
    
    match /dailyThoughts/{thoughtId} {
      allow read: if true;
      allow write: if isAdmin();
    }
    
    match /settings/{settingId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // User personalization collections
    match /users/{userId} {
      allow read: if isOwner(userId) || isAdmin();
      allow write: if isOwner(userId) || isAdmin();
    }
    
    match /bookmarks/{bookmarkId} {
      allow read, write: if isAuthenticated() && resource.data.userId == request.auth.uid;
    }
    
    match /readingHistory/{historyId} {
      allow read, write: if isAuthenticated() && resource.data.userId == request.auth.uid;
    }
  }
}

// =========================================================================
// STORAGE SECURITY RULES (storage.rules)
// =========================================================================

service firebase.storage {
  match /b/{bucket}/o {
    
    function isAdmin() {
      return request.auth != null && 
        (request.auth.token.role == 'admin' || request.auth.token.role == 'superadmin');
    }
    
    function isImage() {
      return request.resource.contentType.matches('image/.*') &&
             request.resource.size < 10 * 1024 * 1024; // Max 10MB
    }
    
    function isPDF() {
      return request.resource.contentType == 'application/pdf' &&
             request.resource.size < 50 * 1024 * 1024; // Max 50MB
    }
    
    function isAudio() {
      return request.resource.contentType.matches('audio/.*') &&
             request.resource.size < 30 * 1024 * 1024; // Max 30MB
    }

    // Public Assets
    match /assets/{allPaths=**} {
      allow read: if true;
      allow write: if isAdmin() && (isImage() || isPDF() || isAudio());
    }
    
    // User Profile Pictures
    match /users/{userId}/{allPaths=**} {
      allow read: if true;
      allow write: if request.auth != null && request.auth.uid == userId && isImage();
    }
  }
}
